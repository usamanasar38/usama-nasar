export const Counter = () => {
  const [count, setCount] = React.useState(0);
  return (
    <button type="button" className="inline-flex h-9 items-center rounded-full border px-4 text-small transition-colors hover:bg-muted" onClick={() => setCount((c) => c + 1)}>
      Clicked {count} {count === 1 ? "time" : "times"}
    </button>
  );
};

This post is written in **MDX**, so it can render components inline:

<Counter />

## Code

```tsx title="counter.tsx"
export const Counter = () => {
  const [count, setCount] = useState(0);
  return <button onClick={() => setCount((c) => c + 1)}>{count}</button>;
};
```
